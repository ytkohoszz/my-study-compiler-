import esbuild from "esbuild";
import process from "process";
import builtins from "builtin-modules";

const prod = process.argv[2] === "production";

const context = await esbuild.context({
  entryPoints: ["src/main.ts"],
  bundle: true,
  external: [
    "obsidian",
    "electron",
    "@codemirror/autocomplete",
    "@codemirror/collab",
    "@codemirror/commands",
    "@codemirror/language",
    "@codemirror/lint",
    "@codemirror/search",
    "@codemirror/state",
    "@codemirror/view",
    "@lezer/common",
    "@lezer/highlight",
    "@lezer/lr",
    ...builtins,
  ],
  format: "cjs",
  target: "es2018",
  logLevel: "info",
  sourcemap: prod ? false : "inline",
  treeShaking: true,
  minify: prod,
  outfile: "main.js",
});

if (prod) {
  await context.rebuild();
  process.exit(0);
} else {
  await context.watch();
}

export function byName(a: StudyNoteItem, b: StudyNoteItem): number {
  return a.name.localeCompare(b.name, "ru", { numeric: true });
}

export function sortActiveRoots(a: StudyNoteItem, b: StudyNoteItem): number {
  const dA = a.deadline ?? "9999-99-99";
  const dB = b.deadline ?? "9999-99-99";
  if (dA !== dB) return dA.localeCompare(dB);
  return byName(a, b);
}

export function sortCompletedRoots(a: StudyNoteItem, b: StudyNoteItem): number {
  if (a.finishedAt && b.finishedAt && a.finishedAt !== b.finishedAt) {
    return b.finishedAt.localeCompare(a.finishedAt);
  }
  return byName(a, b);
}

export function normalizeSpaces(s: string): string {
  return String(s ?? "").replace(/\s+/g, " ").trim();
}

function stripOuterWikiBrackets(str: string): string {
  let s = String(str || "").trim();
  if (s.startsWith("[[")) s = s.slice(2);
  if (s.endsWith("]]")) s = s.slice(0, -2);
  return s;
}

export function flattenFrontmatterValue(val: unknown): string[] {
  if (val === null || val === undefined) return [];
  if (Array.isArray(val)) {
    const res: string[] = [];
    for (const item of val) res.push(...flattenFrontmatterValue(item));
    return res;
  }
  if (typeof val === "object" && (val as { path?: unknown }).path) {
    return [String((val as { path: unknown }).path).trim()];
  }
  const s = String(val).trim();
  return s ? [s] : [];
}

function readString(v: unknown): string {
  return v === null || v === undefined ? "" : String(v).trim();
}

export function isRootStatusCompleted(status: string): boolean {
  const clean = normalizeSpaces(status).toLowerCase();
  if (!clean) return false;
  return ROOT_DONE_STATUSES.includes(clean);
}

export function resolveParentPaths(
  app: App,
  file: TFile,
  rawParentProp: unknown
): string[] {
  const resolved: string[] = [];
  for (const raw of flattenFrontmatterValue(rawParentProp)) {
    const clean = stripOuterWikiBrackets(raw).split("|")[0]?.split("#")[0]?.trim();
    if (!clean) continue;

    const direct = app.vault.getAbstractFileByPath(clean);
    if (direct instanceof TFile) {
      if (!resolved.includes(direct.path)) resolved.push(direct.path);
      continue;
    }
    const withMd = app.vault.getAbstractFileByPath(
      clean.endsWith(".md") ? clean : `${clean}.md`
    );
    if (withMd instanceof TFile) {
      if (!resolved.includes(withMd.path)) resolved.push(withMd.path);
      continue;
    }
    const dest = app.metadataCache.getFirstLinkpathDest(clean, file.path);
    if (dest instanceof TFile && !resolved.includes(dest.path)) {
      resolved.push(dest.path);
    }
  }
  return resolved;
}

export function todayISO(): string {
  const d = new Date();
  const p = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
}

export function daysUntilISO(iso: string): number {
  const parts = iso.split("-").map(Number);
  const y = parts[0];
  const m = parts[1];
  const d = parts[2];
  if (!y || !m || !d) return 999;

  const now = new Date();
  const todayLocalMidnight = new Date(
    now.getFullYear(),
    now.getMonth(),
    now.getDate()
  );
  const targetLocalMidnight = new Date(y, m - 1, d);
  return Math.round(
    (targetLocalMidnight.getTime() - todayLocalMidnight.getTime()) / 86_400_000
  );
}

export function pluralizeDays(n: number): string {
  const abs = Math.abs(Math.trunc(n));
  const mod100 = abs % 100;
  const mod10 = abs % 10;
  if (mod100 >= 11 && mod100 <= 19) return `${abs} дней`;
  if (mod10 === 1) return `${abs} день`;
  if (mod10 >= 2 && mod10 <= 4) return `${abs} дня`;
  return `${abs} дней`;
}

export function getDeadlineTone(item: StudyNoteItem): DeadlineTone {
  if (!item.deadline || item.isCompletedRoot) return "none";
  const diff = daysUntilISO(item.deadline);
  if (diff < 0) return "overdue";
  if (diff <= 7) return "soon";
  return "far";
}

export function formatDeadlineBadge(item: StudyNoteItem): string {
  if (!item.deadline) return "";
  if (item.isCompletedRoot) {
    return `${item.deadline.slice(8, 10)}.${item.deadline.slice(5, 7)}.${item.deadline.slice(0, 4)}`;
  }
  const diff = daysUntilISO(item.deadline);
  if (diff === 0) return "сегодня";
  if (diff === 1) return "завтра";
  if (diff > 1) return `осталось ${pluralizeDays(diff)}`;
  if (diff === -1) return "просрочено на 1 день";
  return `просрочено на ${pluralizeDays(Math.abs(diff))}`;
}

export function formatFullDateTooltip(iso: string): string {
  const parts = iso.split("-");
  if (parts.length !== 3) return iso;
  return `Дедлайн: ${parts[2]}.${parts[1]}.${parts[0]}`;
}

function isTemplateOrExcludedFile(
  file: TFile,
  fm: Record<string, unknown>,
  tagsLower: string[],
  settings: StudyCompilerSettings
): boolean {
  const excludedFolders = settings.excludedFolders
    .map((f) => f.trim().replace(/^\/+|\/+$/g, "").toLowerCase())
    .filter((f) => f.length > 0);

  const pathLower = file.path.toLowerCase();
  const folderSegments = pathLower.split("/").slice(0, -1);
  for (const exFolder of excludedFolders) {
    if (
      pathLower.startsWith(`${exFolder}/`) ||
      folderSegments.includes(exFolder)
    ) {
      return true;
    }
  }

  const prefixes = settings.excludedTitlePrefixes
    .map((p) => p.trim().toLowerCase())
    .filter((p) => p.length > 0);
  const baseLower = file.basename.toLowerCase();
  if (prefixes.some((pref) => baseLower.startsWith(pref))) {
    return true;
  }

  const excludedTags = settings.excludedTags
    .map((t) => t.trim().replace(/^#/, "").toLowerCase())
    .filter((t) => t.length > 0);
  if (tagsLower.some((t) => excludedTags.includes(t))) {
    return true;
  }

  const noteTypeLower = readString(fm[KEYS.type]).toLowerCase();
  if (noteTypeLower === "шаблон" || noteTypeLower === "template") {
    return true;
  }
  if (
    fm[KEYS.templateFlag] === true ||
    String(fm[KEYS.templateFlag]).toLowerCase() === "true"
  ) {
    return true;
  }

  return false;
}

const SYSTEM_FM_IGNORE = new Set([
  KEYS.area,

  "position",

  "alias",
  "cssclasses",

]);

export function scanStudyVault(
  app: App,
  settings: StudyCompilerSettings
): ScanResult {
  const byPath = new Map<string, StudyNoteItem>();
  const typeCounter = new Map<string, { display: string; count: number }>();
  const propKeyFreq = new Map<string, number>();
  const valuesSetByKey = new Map<string, Set<string>>();

  for (const file of app.vault.getMarkdownFiles()) {
    const cache = app.metadataCache.getFileCache(file);
    const fm = cache?.frontmatter;
    if (!fm) continue;

    if (!flattenFrontmatterValue(fm[KEYS.area]).includes(settings.studyAreaName)) {
      continue;
    }

    const isCardDeck =
      Boolean(fm["колода_карточек"]) ||
      readString(fm["подтип"]).toLowerCase() === "карточки";
    if (isCardDeck) continue;

    const rawTags = cache ? getAllTags(cache) ?? [] : [];
    const tags = rawTags.map((t) => t.replace(/^#/, "").toLowerCase());

    if (isTemplateOrExcludedFile(file, fm, tags, settings)) {
      continue;
    }

    const noteType = normalizeSpaces(readString(fm[KEYS.type]));
    if (noteType) {
      const lowerT = noteType.toLowerCase();
      const prev = typeCounter.get(lowerT);
      if (prev) prev.count++;
      else typeCounter.set(lowerT, { display: noteType, count: 1 });
    }

    for (const [k, rawVal] of Object.entries(fm)) {
      if (SYSTEM_FM_IGNORE.has(k)) continue;
      propKeyFreq.set(k, (propKeyFreq.get(k) ?? 0) + 1);
      const flatVals = flattenFrontmatterValue(rawVal);
      if (flatVals.length > 0) {
        let vSet = valuesSetByKey.get(k);
        if (!vSet) {
          vSet = new Set<string>();
          valuesSetByKey.set(k, vSet);
        }
        if (vSet.size < 40) {
          for (const fv of flatVals) {
            if (fv.length <= 80) vSet.add(fv);
          }
        }
      }
    }

    const status = normalizeSpaces(readString(fm[KEYS.status]));
    const rawDeadline = (
      readString(fm[KEYS.deadline]) || readString(fm[KEYS.deadlineAlt])
    ).slice(0, 10);
    const rawFinished = readString(fm[KEYS.finishedAt]).slice(0, 10);

    byPath.set(file.path, {
      file,
      path: file.path,
      name: file.basename,
      noteType,
      status,
      isCompletedRoot: isRootStatusCompleted(status),
      finishedAt: /^\d{4}-\d{2}-\d{2}$/.test(rawFinished) ? rawFinished : null,
      discipline: readString(fm[KEYS.discipline]),
      deadline: /^\d{4}-\d{2}-\d{2}$/.test(rawDeadline) ? rawDeadline : null,
      icon: readString(fm[KEYS.icon]),
      tags,
      parentPaths: resolveParentPaths(app, file, fm[KEYS.parent]),
    });
  }

  const childrenOf = new Map<string, string[]>();

  for (const item of byPath.values()) {
    const parents = item.parentPaths.filter((p) => p !== item.path && byPath.has(p));
    item.parentPaths = parents;
    for (const p of parents) {
      const list = childrenOf.get(p) ?? [];
      if (!list.includes(item.path)) list.push(item.path);
      childrenOf.set(p, list);
    }
  }

  const roots: StudyNoteItem[] = [];
  const orphans: StudyNoteItem[] = [];
  const legacyMaterialsLower = new Set(
    LEGACY_MATERIAL_TYPES.map((t) => t.toLowerCase())
  );

  for (const item of byPath.values()) {
    if (item.parentPaths.length > 0) continue;

    const hasChildren = (childrenOf.get(item.path)?.length ?? 0) > 0;
    const hasRootMarkers = Boolean(item.status || item.deadline);
    const isLegacyLeafMaterial =
      !hasChildren &&
      !hasRootMarkers &&
      legacyMaterialsLower.has(item.noteType.toLowerCase());

    if (isLegacyLeafMaterial) {
      orphans.push(item);
    } else {
      roots.push(item);
    }
  }

  const activeRoots = roots
    .filter((r) => !r.isCompletedRoot)
    .sort(sortActiveRoots);
  const completedRoots = roots
    .filter((r) => r.isCompletedRoot)
    .sort(sortCompletedRoots);
  roots.sort(byName);
  orphans.sort(byName);

  const typeCounts: TypeSuggestion[] = [...typeCounter.values()]
    .map((x) => ({ name: x.display, count: x.count }))
    .sort((a, b) => b.count - a.count || a.name.localeCompare(b.name, "ru"));

  const propertyKeys = [...propKeyFreq.entries()]
    .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0], "ru"))
    .map(([k]) => k);

  const valuesByKey = new Map<string, string[]>();
  for (const [k, set] of valuesSetByKey.entries()) {
    valuesByKey.set(
      k,
      [...set].sort((a, b) => a.localeCompare(b, "ru", { numeric: true }))
    );
  }

  return {
    roots,
    activeRoots,
    completedRoots,
    orphans,
    byPath,
    childrenOf,
    schema: { typeCounts, propertyKeys, valuesByKey },
  };
}

export function findCaseInsensitiveTypeMatch(
  input: string,
  suggestions: TypeSuggestion[]
): string | null {
  const clean = normalizeSpaces(input);
  if (!clean) return null;
  const lower = clean.toLowerCase();
  for (const s of suggestions) {
    if (s.name.toLowerCase() === lower && s.name !== clean) {
      return s.name;
    }
  }
  return null;
}

/**
 * Преобразует строковый ввод произвольного свойства в корректный тип для YAML:
 * - пустая строка -> undefined (не записывать)
 * - kind === "list" -> массив строк по запятой
 * - YYYY-MM-DD -> строка (будет записана в кавычках в YAML)
 * - true / false -> boolean
 * - число -> number без кавычек
 */
export function parseCustomPropertyValue(
  rawValue: string,
  kind: CustomPropValueKind
): unknown {
  const trimmed = rawValue.trim();
  if (!trimmed) return undefined;

  if (kind === "list") {
    const items = trimmed
      .split(",")
      .map((x) => normalizeSpaces(x))
      .filter((x) => x.length > 0);
    return items.length > 0 ? items : undefined;
  }

  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) {
    return trimmed;
  }
  if (trimmed === "true") return true;
  if (trimmed === "false") return false;

  if (/^-?\d+(?:\.\d+)?$/.test(trimmed)) {
    const num = Number(trimmed);
    if (Number.isFinite(num)) return num;
  }

  return trimmed;
}

export function serializeYamlScalarOrList(val: unknown): string {
  if (Array.isArray(val)) {
    return `[${val.map((x) => JSON.stringify(String(x))).join(", ")}]`;
  }
  if (typeof val === "number" || typeof val === "boolean") {
    return String(val);
  }
  return JSON.stringify(String(val ?? ""));
}

export function isReservedYamlKey(key: string): boolean {
  const low = normalizeSpaces(key).toLowerCase();
  return RESERVED_YAML_KEYS.some((k) => k.toLowerCase() === low);
}

export function findRootPath(scan: ScanResult, path: string): string {
  const seen = new Set<string>();
  let cur = path;
  for (;;) {
    if (seen.has(cur)) return cur;
    seen.add(cur);
    const next = scan.byPath.get(cur)?.parentPaths[0];
    if (!next) return cur;
    cur = next;
  }
}

export function buildBreadcrumbChain(
  scan: ScanResult,
  projectRootPath: string,
  focusPath: string
): StudyNoteItem[] {
  const chain: StudyNoteItem[] = [];
  const seen = new Set<string>();
  let cur: string | undefined = focusPath;

  while (cur && !seen.has(cur)) {
    seen.add(cur);
    const item = scan.byPath.get(cur);
    if (!item) break;
    chain.unshift(item);
    if (cur === projectRootPath) break;
    cur = item.parentPaths[0];
  }

  const rootItem = scan.byPath.get(projectRootPath);
  if (rootItem && chain[0]?.path !== projectRootPath) {
    chain.unshift(rootItem);
  }
  return chain;
}

export function collectDescendants(path: string, scan: ScanResult): Set<string> {
  const result = new Set<string>();
  const stack: string[] = [path];
  while (stack.length > 0) {
    const cur = stack.pop();
    if (cur === undefined) break;
    for (const kid of scan.childrenOf.get(cur) ?? []) {
      if (kid === path || result.has(kid)) continue;
      result.add(kid);
      stack.push(kid);
    }
  }
  return result;
}

function matchesFilter(item: StudyNoteItem, filter: TreeFilterOptions): boolean {
  if (
    filter.typeFilter &&
    item.noteType.toLowerCase() !== filter.typeFilter.toLowerCase()
  ) {
    return false;
  }

  const q = filter.query.trim().toLowerCase();
  if (!q) return true;

  const tokens = q.split(/\s+/).filter(Boolean);
  const haystack = [
    item.name,
    item.noteType,
    item.discipline,
    ...item.tags.map((t) => `#${t}`),
  ]
    .join(" ")
    .toLowerCase();

  return tokens.every((tok) => haystack.includes(tok));
}

export function buildNestedFilteredTree(
  branchRoot: StudyNoteItem,
  scan: ScanResult,
  expanded: Set<string> | null,
  filter: TreeFilterOptions
): TreeNode[] {
  if (filter.mode === "orphans") {
    return scan.orphans
      .filter((it) => matchesFilter(it, { ...filter, mode: "all" }))
      .map((item) => ({
        item,
        depth: 0,
        hasChildren: false,
        isExpanded: false,
        directChildrenCount: 0,
        descendantCount: 0,
        isDirectMatch: true,
        children: [],
      }));
  }

  const isFiltering =
    filter.query.trim().length > 0 || filter.typeFilter.length > 0;

  const directMatches = new Set<string>();
  const visibleBranchPaths = new Set<string>();

  if (isFiltering) {
    const checkSubtree = (path: string, trail: Set<string>): boolean => {
      let anyChildMatched = false;
      for (const kidPath of scan.childrenOf.get(path) ?? []) {
        if (trail.has(kidPath)) continue;
        const kid = scan.byPath.get(kidPath);
        if (!kid) continue;
        trail.add(kidPath);
        const childHas = checkSubtree(kidPath, trail);
        trail.delete(kidPath);
        const selfMatch = matchesFilter(kid, filter);
        if (selfMatch) directMatches.add(kidPath);
        if (selfMatch || childHas) {
          visibleBranchPaths.add(kidPath);
          anyChildMatched = true;
        }
      }
      return anyChildMatched;
    };
    checkSubtree(branchRoot.path, new Set([branchRoot.path]));
  }

  const buildLevel = (
    parentPath: string,
    depth: number,
    trail: Set<string>
  ): TreeNode[] => {
    const kids = (scan.childrenOf.get(parentPath) ?? [])
      .map((p) => scan.byPath.get(p))
      .filter((x): x is StudyNoteItem => x !== undefined)
      .sort(byName);

    const levelNodes: TreeNode[] = [];

    for (const kid of kids) {
      if (trail.has(kid.path)) continue;
      if (isFiltering && !visibleBranchPaths.has(kid.path)) continue;

      const allKidChildren = scan.childrenOf.get(kid.path) ?? [];
      const hasChildren = isFiltering
        ? allKidChildren.some((k) => visibleBranchPaths.has(k))
        : allKidChildren.length > 0;

      const descendants = collectDescendants(kid.path, scan);
      const isExpanded =
        hasChildren &&
        (isFiltering || expanded === null || expanded.has(kid.path));

      let childrenNodes: TreeNode[] = [];
      if (isExpanded) {
        trail.add(kid.path);
        childrenNodes = buildLevel(kid.path, depth + 1, trail);
        trail.delete(kid.path);
      }

      levelNodes.push({
        item: kid,
        depth,
        hasChildren,
        isExpanded,
        directChildrenCount: allKidChildren.length,
        descendantCount: descendants.size,
        isDirectMatch: isFiltering ? directMatches.has(kid.path) : false,
        children: childrenNodes,
      });
    }

    return levelNodes;
  };

  return buildLevel(branchRoot.path, 0, new Set([branchRoot.path]));
}

export function flattenVisibleTree(nodes: TreeNode[]): TreeNode[] {
  const flat: TreeNode[] = [];
  const walk = (list: TreeNode[]) => {
    for (const n of list) {
      flat.push(n);
      if (n.isExpanded && n.children.length > 0) {
        walk(n.children);
      }
    }
  };
  walk(nodes);
  return flat;
}

function pickStableHashColor(str: string): string {
  let h = 2166136261;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  const idx = Math.abs(h) % HASH_PALETTE.length;
  return HASH_PALETTE[idx] ?? "var(--color-blue, #599ed6)";
}

export async function ensureFolder(app: App, folderPath: string): Promise<void> {
  let current = "";
  for (const part of normalizePath(folderPath).split("/").filter(Boolean)) {
    current = current ? `${current}/${part}` : part;
    if (!app.vault.getAbstractFileByPath(current)) {
      try {
        await app.vault.createFolder(current);
      } catch {
        // папка уже создана параллельно
      }
    }
  }
}

  private rebuild(): void {
    const scan = this.scan;
    const root = this.branchRoot;
    if (!scan || !root) return;

    const visited = new Set<string>([root.path]);
    const extra: Array<[string, string]> = [];

    const make = (item: StudyNoteItem, depth: number): LayoutNode => {
      const kidPaths = scan.childrenOf.get(item.path) ?? [];
      const node: LayoutNode = {
        item,
        depth,
        x: 0,
        y: 0,
        children: [],
        hiddenCount: 0,
        hasKids: kidPaths.length > 0,
      };
      if (!node.hasKids) return node;
      if (this.collapsed.has(item.path)) {
        node.hiddenCount = collectDescendants(item.path, scan).size;
        return node;
      }
      const kids = kidPaths
        .map((p) => scan.byPath.get(p))
        .filter((x): x is StudyNoteItem => x !== undefined)
        .sort(byName);
      for (const kid of kids) {
        if (visited.has(kid.path)) {
          extra.push([item.path, kid.path]);
          continue;
        }
        visited.add(kid.path);
        node.children.push(make(kid, depth + 1));
      }
      return node;
    };

    const tree = make(root, 0);

    let leafIndex = 0;
    const place = (n: LayoutNode) => {
      n.x = n.depth * COL_W;
      if (n.children.length === 0) {
        n.y = leafIndex * ROW_H;
        leafIndex += 1;
        return;
      }
      for (const c of n.children) place(c);
      const first = n.children[0];
      const last = n.children[n.children.length - 1];
      n.y = first && last ? (first.y + last.y) / 2 : 0;
    };
    place(tree);

    this.placed = [];
    this.nodeByPath.clear();
    this.parentOf.clear();
    const stack: LayoutNode[] = [tree];
    while (stack.length > 0) {
      const n = stack.pop();
      if (!n) break;
      this.placed.push(n);
      this.nodeByPath.set(n.item.path, n);
      for (const c of n.children) {
        this.parentOf.set(c.item.path, n.item.path);
        stack.push(c);
      }
    }

    let maxX = 0;
    let maxY = 0;
    for (const n of this.placed) {
      maxX = Math.max(maxX, n.x + NODE_W + 24);
      maxY = Math.max(maxY, n.y + NODE_H);
    }
    this.bounds = { minX: 0, minY: 0, w: Math.max(maxX, 1), h: Math.max(maxY, 1) };

    this.renderLayers(extra);

    if (this.selectedPath && !this.nodeByPath.has(this.selectedPath)) {
      this.selectedPath = null;
    }
    this.applySelection();
    this.updateInfo();
    this.syncToggleAll();

    let hiddenTotal = 0;
    for (const n of this.placed) hiddenTotal += n.hiddenCount;
    this.statEl.setText(
      hiddenTotal > 0
        ? `Узлов: ${this.placed.length} · скрыто: ${hiddenTotal}`
        : `Узлов: ${this.placed.length}`
    );
  }

  private select(path: string | null): void {
    this.selectedPath = path;
    this.applySelection();
    this.updateInfo();
  }

  private lineageOf(path: string): Set<string> {
    const set = new Set<string>([path]);
    let cur = this.parentOf.get(path);
    while (cur !== undefined && !set.has(cur)) {
      set.add(cur);
      cur = this.parentOf.get(cur);
    }
    const node = this.nodeByPath.get(path);
    const stack: LayoutNode[] = node ? [...node.children] : [];
    while (stack.length > 0) {
      const n = stack.pop();
      if (!n) break;
      set.add(n.item.path);
      for (const c of n.children) stack.push(c);
    }
    return set;
  }

  private applySelection(): void {
    const sel = this.selectedPath;
    const lineage = sel ? this.lineageOf(sel) : null;
    for (const [path, el] of this.nodeEls) {
      el.classList.toggle("is-selected", path === sel);
      el.classList.toggle("is-dim", lineage !== null && !lineage.has(path));
    }
    for (const e of this.edgeRefs) {
      const lit = lineage !== null && lineage.has(e.from) && lineage.has(e.to);
      e.el.classList.toggle("is-lit", lit);
      e.el.classList.toggle("is-dim", lineage !== null && !lit);
    }
  }

  private updateInfo(): void {
    const node = this.selectedPath ? this.nodeByPath.get(this.selectedPath) : undefined;
    if (!node) {
      this.infoEl.addClass("is-hidden");
      return;
    }
    const it = node.item;
    const meta = getWorkMeta(it.noteType, it.icon);
    this.infoEl.removeClass("is-hidden");
    this.infoTitleEl.setText(`${meta.icon} ${it.name}`);
    const parts = [it.noteType || "без типа"];
    if (it.deadline) parts.push(formatDeadlineBadge(it));
    this.infoSubEl.setText(parts.join(" · "));
    this.focusBtn.disabled = it.path === this.branchRoot?.path;
  }

  // ---------- Масштаб и сдвиг ----------

  private tryFit(): void {
    if (this.svg.clientWidth <= 0 || this.svg.clientHeight <= 0) return;
    this.pendingFit = false;
    this.fit(false);
  }

  fit(animate: boolean): void {
    const w = this.svg.clientWidth;
    const h = this.svg.clientHeight;
    if (w <= 0 || h <= 0) return;
    const pad = 28;
    const b = this.bounds;
    const wFit = (w - pad * 2) / b.w;
    const hFit = (h - pad * 2) / b.h;
    let k = Math.min(wFit, hFit, 1.1);
    k = Math.max(k, Math.min(0.55, wFit));
    this.k = clamp(k, MIN_K, MAX_K);
    this.tx =
      b.w * this.k + pad * 2 <= w
        ? (w - b.w * this.k) / 2 - b.minX * this.k
        : pad - b.minX * this.k;
    this.ty =
      b.h * this.k + pad * 2 <= h
        ? (h - b.h * this.k) / 2 - b.minY * this.k
        : pad - b.minY * this.k;
    this.applyTransform(animate);
  }

  private applyTransform(animate = false): void {
    const vp = this.viewport;
    if (animate && !prefersReducedMotion()) {
      vp.classList.add("is-animating");
      if (this.animTimer !== null) window.clearTimeout(this.animTimer);
      this.animTimer = window.setTimeout(() => {
        vp.classList.remove("is-animating");
        this.animTimer = null;
      }, 260);
    }
    vp.style.transform = `translate(${this.tx}px, ${this.ty}px) scale(${this.k})`;
  }

  private zoomAt(clientX: number, clientY: number, factor: number, animate = false): void {
    const rect = this.svg.getBoundingClientRect();
    const px = clientX - rect.left;
    const py = clientY - rect.top;
    const nextK = clamp(this.k * factor, MIN_K, MAX_K);
    const ratio = nextK / this.k;
    this.tx = px - (px - this.tx) * ratio;
    this.ty = py - (py - this.ty) * ratio;
    this.k = nextK;
    this.applyTransform(animate);
  }

  private zoomByCenter(factor: number): void {
    const rect = this.svg.getBoundingClientRect();
    this.zoomAt(rect.left + rect.width / 2, rect.top + rect.height / 2, factor, true);
  }

  private stopAnimation(): void {
    this.viewport.classList.remove("is-animating");
  }

  // ---------- Указатель ----------

  private onWheel = (e: WheelEvent): void => {
    e.preventDefault();
    this.stopAnimation();
    const dy = clamp(e.deltaY, -120, 120);
    this.zoomAt(e.clientX, e.clientY, Math.exp(-dy * 0.002));
  };

  private onPointerDown = (e: PointerEvent): void => {
    if (e.pointerType === "mouse" && e.button !== 0) return;
    this.stopAnimation();
    this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
    if (this.pointers.size === 1) {
      this.downPos = { x: e.clientX, y: e.clientY };
      this.downTarget = e.target instanceof Element ? e.target : null;
      this.moved = false;
      this.attachDocListeners();
    } else {
      this.moved = true;
    }
  };

  private onPointerMove = (e: PointerEvent): void => {
    const prev = this.pointers.get(e.pointerId);
    if (!prev) return;
    const cur: Point = { x: e.clientX, y: e.clientY };

    if (this.pointers.size === 1) {
      if (
        !this.moved &&
        Math.hypot(cur.x - this.downPos.x, cur.y - this.downPos.y) > DRAG_THRESHOLD
      ) {
        this.moved = true;
      }
      if (this.moved) {
        this.tx += cur.x - prev.x;
        this.ty += cur.y - prev.y;
        this.applyTransform();
      }
      this.pointers.set(e.pointerId, cur);
      return;
    }

    const before = this.metrics();
    this.pointers.set(e.pointerId, cur);
    const after = this.metrics();
    if (before && after && before.dist > 0) {
      this.tx += after.cx - before.cx;
      this.ty += after.cy - before.cy;
      this.zoomAt(after.cx, after.cy, after.dist / before.dist);
    }
  };

  private onPointerUp = (e: PointerEvent): void => {
    if (!this.pointers.has(e.pointerId)) return;
    this.pointers.delete(e.pointerId);
    if (this.pointers.size === 0) {
      this.detachDocListeners();
      const target = this.downTarget;
      this.downTarget = null;
      if (!this.moved && e.type === "pointerup") this.handleTap(target);
    }
  };

  private metrics(): { cx: number; cy: number; dist: number } | null {
    const pts = [...this.pointers.values()];
    const a = pts[0];
    const b = pts[1];
    if (!a || !b) return null;
    return {
      cx: (a.x + b.x) / 2,
      cy: (a.y + b.y) / 2,
      dist: Math.hypot(a.x - b.x, a.y - b.y),
    };
  }

  private attachDocListeners(): void {
    if (this.listening) return;
    this.listening = true;
    this.doc.addEventListener("pointermove", this.onPointerMove);
    this.doc.addEventListener("pointerup", this.onPointerUp);
    this.doc.addEventListener("pointercancel", this.onPointerUp);
  }

  private detachDocListeners(): void {
    if (!this.listening) return;
    this.listening = false;
    this.doc.removeEventListener("pointermove", this.onPointerMove);
    this.doc.removeEventListener("pointerup", this.onPointerUp);
    this.doc.removeEventListener("pointercancel", this.onPointerUp);
  }

  private handleTap(target: Element | null): void {
    if (!target) return;

    const toggle = target.closest("[data-toggle]");
    if (toggle) {
      const p = toggle.getAttribute("data-toggle");
      if (p) this.toggleCollapse(p);
      return;
    }

    const nodeEl = target.closest("[data-path]");
    const path = nodeEl ? nodeEl.getAttribute("data-path") : null;
    if (!path) {
      this.select(null);
      return;
    }

    const now = Date.now();
    if (this.lastTapPath === path && now - this.lastTapAt < DOUBLE_TAP_MS) {
      this.lastTapPath = null;
      this.callbacks.onOpen(path);
      return;
    }
    this.lastTapPath = path;
    this.lastTapAt = now;
    this.select(path);
  }
}

const DURATION_MS = 180;

export function prefersReducedMotion(): boolean {
  return window.matchMedia("(prefers-reduced-motion: reduce)").matches;
}

export function animateOpen(el: HTMLElement): void {
  if (prefersReducedMotion()) return;
  const height = el.offsetHeight;
  if (height <= 0) return;
  el.style.overflow = "hidden";
  const anim = el.animate(
    [
      { height: "0px", opacity: 0 },
      { height: `${height}px`, opacity: 1 },
    ],
    { duration: DURATION_MS, easing: "ease-out" }
  );
  const done = () => {
    el.style.overflow = "";
  };
  anim.onfinish = done;
  anim.oncancel = done;
}

export function animateClose(el: HTMLElement): Promise<void> {
  if (prefersReducedMotion()) return Promise.resolve();
  const height = el.offsetHeight;
  el.style.overflow = "hidden";
  const anim = el.animate(
    [
      { height: `${height}px`, opacity: 1 },
      { height: "0px", opacity: 0 },
    ],
    { duration: DURATION_MS, easing: "ease-in", fill: "forwards" }
  );
  return anim.finished.then(
    () => undefined,
    () => undefined
  );
}

  private noRestore() {
    (this as unknown as { shouldRestoreSelection?: boolean }).shouldRestoreSelection = false;
  }

  /**
   * Переводит фокус на сам контейнер окна (tabindex="-1"), снимая его с <select>, <input> и <button>.
   * Благодаря этому на iPad не выскакивает системное меню <select> или экранная клавиатура,
   * а на десктопе продолжают работать горячие клавиши (Ctrl/Cmd+F и стрелки).
   */
  private focusSafeContainer() {
    if (this.closed) return;
    const active = document.activeElement;
    if (
      active instanceof HTMLSelectElement ||
      active instanceof HTMLInputElement ||
      active instanceof HTMLTextAreaElement ||
      active instanceof HTMLButtonElement
    ) {
      active.blur();
    }
    this.contentEl.focus({ preventScroll: true });
  }

  onOpen(): void {
    this.noRestore();
    this.closed = false;
    this.viewMode = "tree";
    this.modalEl.addClass("my-sc-modal-container");
    this.contentEl.addClass("my-sc-modal");
    this.contentEl.setAttribute("tabindex", "-1");

    this.datalistsEl = this.contentEl.createDiv("my-sc-datalists-host");
    this.barEl = this.contentEl.createDiv("my-sc-project-bar");
    this.searchWrapEl = this.contentEl.createDiv("my-sc-search-panel");
    this.crumbsEl = this.contentEl.createDiv("my-sc-crumbs-bar");
    this.formEl = this.contentEl.createDiv("my-sc-form-wrap");
    this.treeEl = this.contentEl.createDiv("my-sc-body");
    this.graphEl = this.contentEl.createDiv("my-sc-graph-wrap is-hidden");

    const onKeyDown = (e: KeyboardEvent) => this.handleKeyboardNav(e);
    this.modalEl.addEventListener("keydown", onKeyDown);
    this.cleanups.push(() => this.modalEl.removeEventListener("keydown", onKeyDown));

    const onChange = () => this.scheduleRefresh();
    const r1 = this.app.metadataCache.on("changed", onChange);
    const r2 = this.app.vault.on("delete", onChange);
    const r3 = this.app.vault.on("rename", onChange);
    this.cleanups.push(
      () => this.app.metadataCache.offref(r1),
      () => this.app.vault.offref(r2),
      () => this.app.vault.offref(r3)
    );

    this.scan = scanStudyVault(this.app, this.plugin.settings);
    this.knownPaths = new Set(this.scan.byPath.keys());
    const addUnder = this.options.addUnder;
    if (addUnder && this.scan.byPath.has(addUnder)) {
      this.selectedPath = findRootPath(this.scan, addUnder);
    }

    this.syncGroupTabWithSelectedProject();

    const root = this.getRootInCurrentGroup();
    if (root) {
      const savedFocus = this.plugin.settings.focusedBranchByWork[root.path];
      if (savedFocus && this.scan.byPath.has(savedFocus)) {
        this.focusBranchPath = savedFocus;
      }
    }

    this.renderDatalists();
    this.renderBar();
    this.renderSearchPanel();
    this.renderBreadcrumbs();
    this.applyViewMode();
    this.renderTree();

    if (addUnder && this.scan.byPath.has(addUnder)) {
      // Временно блокируем автофокус на <select>, пока через 20 мс фокус не перейдёт в поле названия формы
      if (this.workSelectEl) this.workSelectEl.tabIndex = -1;
      this.openForm(addUnder, true);
      const restoreSelectTabIdx = window.setTimeout(() => {
        if (this.workSelectEl) this.workSelectEl.removeAttribute("tabindex");
      }, 60);
      this.cleanups.push(() => window.clearTimeout(restoreSelectTabIdx));
    } else {
      // Блокируем перехват фокуса первым <select> со стороны Modal.open() в Obsidian на iPadOS
      if (this.workSelectEl) this.workSelectEl.tabIndex = -1;
      this.focusSafeContainer();
      const t1 = window.setTimeout(() => {
        this.focusSafeContainer();
      }, 0);
      const t2 = window.setTimeout(() => {
        if (this.workSelectEl) this.workSelectEl.removeAttribute("tabindex");
        if (document.activeElement === this.workSelectEl) {
          this.focusSafeContainer();
        }
      }, 60);
      this.cleanups.push(
        () => window.clearTimeout(t1),
        () => window.clearTimeout(t2)
      );
    }
  }

  onClose(): void {
    this.closed = true;
    for (const fn of this.cleanups) fn();
    this.cleanups = [];
    if (this.refreshTimer !== null) {
      window.clearTimeout(this.refreshTimer);
      this.refreshTimer = null;
    }
    if (this.freshTimer !== null) {
      window.clearTimeout(this.freshTimer);
      this.freshTimer = null;
    }
    this.freshPaths.clear();
    if (this.graphView) {
      this.graphView.destroy();
      this.graphView = null;
    }
    this.lastGraphRootPath = "";
    this.expandToggleBtn = null;
    this.workSelectEl = null;
    this.searchInputEl = null;
    this.contentEl.empty();
  }

  private syncGroupTabWithSelectedProject() {
    if (!this.selectedPath) return;
    const found = this.scan.roots.find((r) => r.path === this.selectedPath);
    if (!found) return;
    const targetGroup: ProjectGroupTab = found.isCompletedRoot ? "completed" : "active";
    if (this.groupTab !== targetGroup) {
      this.groupTab = targetGroup;
      this.plugin.settings.projectGroupTab = targetGroup;
      this.plugin.requestSaveSettings();
    }
  }

  private getRootsForTab(tab: ProjectGroupTab): StudyNoteItem[] {
    return tab === "completed" ? this.scan.completedRoots : this.scan.activeRoots;
  }

  private getRootInCurrentGroup(): StudyNoteItem | null {
    const groupList = this.getRootsForTab(this.groupTab);
    const direct = groupList.find((r) => r.path === this.selectedPath);
    if (direct) return direct;
    const first = groupList[0] ?? null;
    if (first) {
      this.selectedPath = first.path;
      this.plugin.settings.lastWorkPath = first.path;
      this.plugin.requestSaveSettings();
    }
    return first;
  }

  private resolveRootWithFallback(): StudyNoteItem | null {
    const root = this.getRootInCurrentGroup();
    if (root) return root;
    const other: ProjectGroupTab = this.groupTab === "active" ? "completed" : "active";
    if (this.getRootsForTab(other).length === 0) return null;
    this.groupTab = other;
    this.plugin.settings.projectGroupTab = other;
    this.plugin.requestSaveSettings();
    return this.getRootInCurrentGroup();
  }

  // ---------- Режимы «Дерево / Граф» ----------

  private applyViewMode() {
    const graph = this.viewMode === "graph";
    this.searchWrapEl.toggleClass("is-hidden", graph);
    this.treeEl.toggleClass("is-hidden", graph);
    this.graphEl.toggleClass("is-hidden", !graph);
  }

  private setViewMode(mode: ViewMode) {
    if (this.viewMode === mode) return;
    this.viewMode = mode;
    this.applyViewMode();
    this.renderBar();
    this.renderMain();
    this.focusSafeContainer();
  }

  private renderMain() {
    if (this.viewMode === "graph") this.renderGraph();
    else this.renderTree();
  }

  private renderGraph() {
    const root = this.getRootInCurrentGroup();
    if (!root) {
      if (this.graphView) {
        this.graphView.destroy();
        this.graphView = null;
      }
      this.lastGraphRootPath = "";
      this.graphEl.empty();
      return;
    }
    const branchRoot = this.getActiveBranchRoot(root);

    if (!this.graphView) {
      this.graphView = new ProjectGraphView(this.graphEl, {
        onOpen: (p) => this.openNote(p),
        onFocus: (p) => {
          const r = this.getRootInCurrentGroup();
          if (r) this.setFocusBranch(r, p);
        },
      });
      this.lastGraphRootPath = "";
    }

    const reset = this.lastGraphRootPath !== branchRoot.path;
    this.lastGraphRootPath = branchRoot.path;
    this.graphView.update(this.scan, branchRoot, reset, this.freshPaths);
  }

  private scheduleRefresh() {
    if (this.refreshTimer !== null) window.clearTimeout(this.refreshTimer);
    this.refreshTimer = window.setTimeout(() => {
      this.refreshTimer = null;
      if (this.closed) return;

      const wasSearchFocused = document.activeElement === this.searchInputEl;
      const inFormOrInlineInput =
        document.activeElement instanceof HTMLInputElement &&
        document.activeElement !== this.searchInputEl;

      const next = scanStudyVault(this.app, this.plugin.settings);
      const fresh: string[] = [];
      for (const p of next.byPath.keys()) {
        if (!this.knownPaths.has(p)) fresh.push(p);
      }
      this.knownPaths = new Set(next.byPath.keys());
      this.scan = next;
      if (fresh.length > 0 && fresh.length <= 20) this.markFresh(fresh);

      this.renderDatalists();
      this.renderBar();
      this.renderSearchPanel();
      this.renderBreadcrumbs();
      if (this.form.open && !inFormOrInlineInput) this.renderForm(false);
      this.renderMain();

      if (wasSearchFocused && this.searchInputEl) {
        this.searchInputEl.focus({ preventScroll: true });
      } else if (!inFormOrInlineInput) {
        this.focusSafeContainer();
      }

      if (this.viewMode === "tree") {
        const firstFresh = fresh
          .map((p) => this.rowElementByPath.get(p))
          .find((el): el is HTMLElement => el !== undefined);
        if (firstFresh) {
          firstFresh.scrollIntoView({
            block: "nearest",
            behavior: prefersReducedMotion() ? "auto" : "smooth",
          });
        }
      }
    }, 220);
  }

  private markFresh(paths: string[]) {
    for (const p of paths) this.freshPaths.add(p);
    if (this.freshTimer !== null) window.clearTimeout(this.freshTimer);
    this.freshTimer = window.setTimeout(() => {
      this.freshTimer = null;
      for (const p of this.freshPaths) {
        this.rowElementByPath.get(p)?.removeClass("is-new");
      }
      this.freshPaths.clear();
      this.graphView?.clearFresh();
    }, 2400);
  }

  private getChildrenContainer(path: string): HTMLElement | null {
    const group = this.rowElementByPath.get(path)?.parentElement;
    return group?.querySelector<HTMLElement>(".my-sc-children-container") ?? null;
  }

  private toggleBranch(path: string) {
    if (this.closingBranch) return;

    if (!this.expanded.has(path)) {
      this.expanded.add(path);
      this.saveExpanded();
      this.pendingOpenPath = path;
      this.renderTree();
      return;
    }

    this.expanded.delete(path);
    this.saveExpanded();
    const kids = this.getChildrenContainer(path);
    if (!kids || prefersReducedMotion()) {
      this.renderTree();
      return;
    }
    this.rowElementByPath
      .get(path)
      ?.querySelector(".my-sc-toggle-btn")
      ?.removeClass("is-open");
    this.closingBranch = true;
    void animateClose(kids).then(() => {
      this.closingBranch = false;
      if (!this.closed) this.renderTree();
    });
  }

  private renderDatalists() {
    this.datalistsEl.empty();

    const typeDl = this.datalistsEl.createEl("datalist", {
      attr: { id: "my-sc-types-datalist" },
    });
    for (const t of this.scan.schema.typeCounts) {
      typeDl.createEl("option", {
        attr: { value: t.name, label: `${t.name} (${t.count})` },
      });
    }

    const keysDl = this.datalistsEl.createEl("datalist", {
      attr: { id: "my-sc-prop-keys-datalist" },
    });
    for (const k of this.scan.schema.propertyKeys) {
      keysDl.createEl("option", { attr: { value: k } });
    }
  }

  private getActiveBranchRoot(projectRoot: StudyNoteItem): StudyNoteItem {
    if (this.focusBranchPath) {
      const target = this.scan.byPath.get(this.focusBranchPath);
      if (target && findRootPath(this.scan, target.path) === projectRoot.path) {
        return target;
      }
    }
    this.focusBranchPath = "";
    return projectRoot;
  }

  private setFocusBranch(projectRoot: StudyNoteItem, branchPath: string) {
    this.focusBranchPath = branchPath === projectRoot.path ? "" : branchPath;
    if (this.focusBranchPath) {
      this.plugin.settings.focusedBranchByWork[projectRoot.path] = this.focusBranchPath;
    } else {
      delete this.plugin.settings.focusedBranchByWork[projectRoot.path];
    }
    this.plugin.requestSaveSettings();
    this.activeRowIndex = -1;
    this.renderBar();
    this.renderBreadcrumbs();
    this.renderMain();
    this.focusSafeContainer();
  }

  private saveExpanded() {
    this.plugin.settings.expandedPaths = [...this.expanded];
    this.plugin.requestSaveSettings();
  }

  private handleKeyboardNav(e: KeyboardEvent) {
    if (this.viewMode === "graph") return;

    if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "f") {
      e.preventDefault();
      this.searchInputEl?.focus();
      this.searchInputEl?.select();
      return;
    }

    const activeEl = document.activeElement;
    const inOtherInput =
      (activeEl instanceof HTMLInputElement && activeEl !== this.searchInputEl) ||
      activeEl instanceof HTMLSelectElement ||
      activeEl instanceof HTMLTextAreaElement;
    if (inOtherInput) return;

    const len = this.currentVisibleFlat.length;
    if (len === 0) return;

    if (e.key === "ArrowDown") {
      e.preventDefault();
      this.activeRowIndex = Math.min(len - 1, this.activeRowIndex + 1);
      this.highlightActiveRow();
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      this.activeRowIndex = Math.max(0, this.activeRowIndex - 1);
      this.highlightActiveRow();
    } else if (e.key === "Enter" && this.activeRowIndex >= 0) {
      const picked = this.currentVisibleFlat[this.activeRowIndex];
      if (picked) {
        e.preventDefault();
        this.openNote(picked.item.path);
      }
    } else if (
      (e.key === "ArrowRight" || e.key === "ArrowLeft") &&
      activeEl !== this.searchInputEl &&
      this.activeRowIndex >= 0
    ) {
      const picked = this.currentVisibleFlat[this.activeRowIndex];
      if (!picked || !picked.hasChildren) return;
      const isOpen = this.expanded.has(picked.item.path);
      if (e.key === "ArrowRight" && isOpen) return;
      if (e.key === "ArrowLeft" && !isOpen) return;
      e.preventDefault();
      this.toggleBranch(picked.item.path);
    }
  }

  private highlightActiveRow() {
    for (const el of this.rowElementByPath.values()) {
      el.removeClass("is-kb-active");
    }
    const picked = this.currentVisibleFlat[this.activeRowIndex];
    if (!picked) return;
    const targetEl = this.rowElementByPath.get(picked.item.path);
    if (targetEl) {
      targetEl.addClass("is-kb-active");
      targetEl.scrollIntoView({ block: "nearest" });
    }
  }

  private iconBtn(
    parent: HTMLElement,
    icon: string,
    title: string,
    onClick: () => void,
    opts: { active?: boolean; primary?: boolean } = {}
  ): HTMLButtonElement {
    const btn = parent.createEl("button", {
      cls:
        "my-sc-icon-btn" +
        (opts.active ? " is-active" : "") +
        (opts.primary ? " is-primary" : ""),
      attr: { type: "button", title, "aria-label": title },
    });
    setIcon(btn, icon);
    btn.onclick = onClick;
    return btn;
  }

  private hasOpenBranches(branchPath: string): boolean {
    for (const p of collectDescendants(branchPath, this.scan)) {
      if (this.expanded.has(p) && (this.scan.childrenOf.get(p)?.length ?? 0) > 0) {
        return true;
      }
    }
    return false;
  }

  private syncExpandToggle() {
    const btn = this.expandToggleBtn;
    if (!btn || !btn.isConnected) return;
    const root = this.getRootInCurrentGroup();
    if (!root) return;
    const branch = this.getActiveBranchRoot(root);
    const open = this.hasOpenBranches(branch.path);
    const title = open ? "Свернуть все ветки" : "Развернуть все ветки";
    btn.empty();
    setIcon(btn, open ? "chevrons-down-up" : "chevrons-up-down");
    btn.setAttribute("title", title);
    btn.setAttribute("aria-label", title);
  }

  private openProjectMenu(e: MouseEvent, root: StudyNoteItem, branchRoot: StudyNoteItem) {
    const menu = new Menu();
    menu.addItem((mi) =>
      mi
        .setTitle(
          branchRoot.path === root.path ? "Открыть заметку проекта" : "Открыть заметку ветки"
        )
        .setIcon("file-text")
        .onClick(() => this.openNote(branchRoot.path))
    );
    menu.addSeparator();
    menu.addItem((mi) =>
      mi
        .setTitle(root.isCompletedRoot ? "Вернуть проект в работу" : "Завершить проект")
        .setIcon(root.isCompletedRoot ? "rotate-ccw" : "check")
        .onClick(() => void this.toggleRootProjectStatus(root))
    );
    menu.showAtMouseEvent(e);
  }

  private renderBar() {
    this.barEl.empty();
    this.expandToggleBtn = null;
    this.workSelectEl = null;

    const root = this.resolveRootWithFallback();
    if (!root) {
      this.formEl.empty();
      this.barEl.createDiv({
        cls: "my-sc-empty-group-box",
        text: `Проектов пока нет. Создай заметку с направлением «${this.plugin.settings.studyAreaName}» без родителя.`,
      });
      return;
    }

    const branchRoot = this.getActiveBranchRoot(root);

    // Ряд 1: выбор проекта + кнопка меню проекта
    const topRow = this.barEl.createDiv("my-sc-top-row");
    const sel = topRow.createEl("select", { cls: "dropdown my-sc-work-select" });
    this.workSelectEl = sel;

    const addGroup = (label: string, list: StudyNoteItem[]) => {
      if (list.length === 0) return;
      const group = sel.createEl("optgroup", {
        attr: { label: `${label} · ${list.length}` },
      });
      for (const r of list) {
        const meta = getWorkMeta(r.noteType, r.icon);
        const dlText = r.deadline && !r.isCompletedRoot ? ` · ${formatDeadlineBadge(r)}` : "";
        const o = group.createEl("option", {
          text: `${meta.icon} ${r.name}${dlText}`,
          value: r.path,
        });
        if (r.path === root.path) o.selected = true;
      }
    };
    addGroup("Активные", this.scan.activeRoots);
    addGroup("Завершённые", this.scan.completedRoots);

    sel.onchange = () => {
      const picked = this.scan.roots.find((r) => r.path === sel.value);
      if (!picked) return;
      this.groupTab = picked.isCompletedRoot ? "completed" : "active";
      this.plugin.settings.projectGroupTab = this.groupTab;
      this.selectedPath = picked.path;
      this.plugin.settings.lastWorkPath = picked.path;
      this.focusBranchPath =
        this.plugin.settings.focusedBranchByWork[picked.path] || "";
      this.plugin.requestSaveSettings();
      this.form.open = false;
      this.formEl.empty();
      this.activeRowIndex = -1;
      this.renderBar();
      this.renderBreadcrumbs();
      this.renderMain();
      this.focusSafeContainer();
    };

    const moreBtn = topRow.createEl("button", {
      cls: "my-sc-icon-btn",
      attr: { type: "button", title: "Действия с проектом", "aria-label": "Действия с проектом" },
    });
    setIcon(moreBtn, "more-horizontal");
    moreBtn.onclick = (e) => this.openProjectMenu(e, root, branchRoot);

    // Ряд 2: сводка слева · сегмент «Дерево | Граф» и кнопки справа
    const descendants = collectDescendants(branchRoot.path, this.scan);
    const metaRow = this.barEl.createDiv("my-sc-meta-row");
    const metaInfo = metaRow.createDiv("my-sc-meta-info");
    metaInfo.createSpan({
      cls: "my-sc-summary-text",
      text:
        descendants.size > 0
          ? `Материалов: ${descendants.size}`
          : "Материалов пока нет",
    });
    this.renderDeadline(metaInfo, branchRoot);

    const tools = metaRow.createDiv("my-sc-icon-group");

    const modeSeg = tools.createDiv("my-sc-view-seg");
    const addModeBtn = (mode: ViewMode, icon: string, label: string) => {
      const b = modeSeg.createEl("button", {
        cls: "my-sc-view-seg-btn" + (this.viewMode === mode ? " is-active" : ""),
        attr: {
          type: "button",
          title: mode === "tree" ? "Вид дерева" : "Вид графа",
          "aria-pressed": String(this.viewMode === mode),
        },
      });
      const icSpan = b.createSpan("my-sc-view-seg-icon");
      setIcon(icSpan, icon);
      b.createSpan({ cls: "my-sc-view-seg-label", text: label });
      b.onclick = () => this.setViewMode(mode);
    };
    addModeBtn("tree", "list", "Дерево");
    addModeBtn("graph", "network", "Граф");

    if (this.viewMode === "tree") {
      this.expandToggleBtn = this.iconBtn(tools, "chevrons-up-down", "Развернуть все ветки", () => {
        const open = this.hasOpenBranches(branchRoot.path);
        for (const p of collectDescendants(branchRoot.path, this.scan)) {
          if (open) this.expanded.delete(p);
          else if ((this.scan.childrenOf.get(p)?.length ?? 0) > 0) this.expanded.add(p);
        }
        this.saveExpanded();
        this.renderTree();
        this.focusSafeContainer();
      });
      this.syncExpandToggle();
    }

    this.iconBtn(
      tools,
      this.form.open ? "x" : "plus",
      this.form.open ? "Закрыть форму" : "Добавить материал",
      () => {
        if (this.form.open) {
          this.form.open = false;
          this.formEl.empty();
          this.renderBar();
          this.focusSafeContainer();
        } else {
          this.openForm(branchRoot.path, true);
        }
      },
      { primary: true, active: this.form.open }
    );
  }

  private async toggleRootProjectStatus(root: StudyNoteItem) {
    const nextDone = !root.isCompletedRoot;
    try {
      await this.app.fileManager.processFrontMatter(
        root.file,
        (fm: Record<string, unknown>) => {
          fm[KEYS.status] = nextDone ? ROOT_STATUS_DONE : ROOT_STATUS_ACTIVE;
          if (nextDone && !fm[KEYS.finishedAt]) {
            fm[KEYS.finishedAt] = todayISO();
          }
        }
      );
      this.groupTab = nextDone ? "completed" : "active";
      this.plugin.settings.projectGroupTab = this.groupTab;
      this.plugin.requestSaveSettings();
      new Notice(
        nextDone
          ? `Проект «${root.name}» перенесён в завершённые`
          : `Проект «${root.name}» возвращён в активные`
      );
    } catch (err: unknown) {
      new Notice(
        "Не удалось изменить статус проекта: " +
          (err instanceof Error ? err.message : String(err))
      );
    }
  }

  private renderSearchPanel() {
    this.searchWrapEl.empty();

    const inp = this.searchWrapEl.createEl("input", {
      cls: "my-sc-search-input",
      attr: {
        type: "search",
        placeholder: "Название, тип или #тег",
      },
    });
    inp.value = this.searchQuery;
    this.searchInputEl = inp;
    inp.oninput = () => {
      this.searchQuery = inp.value;
      this.activeRowIndex = 0;
      this.renderTree();
    };

    const orphanCount = this.scan.orphans.length;
    if (orphanCount > 0 || this.filterMode === "orphans") {
      const chip = this.searchWrapEl.createEl("button", {
        cls: "my-sc-filter-chip" + (this.filterMode === "orphans" ? " is-active" : ""),
        text: `Без родителя · ${orphanCount}`,
        attr: { type: "button", title: "Показать материалы, не привязанные к дереву" },
      });
      chip.onclick = () => {
        this.filterMode = this.filterMode === "orphans" ? "all" : "orphans";
        this.activeRowIndex = -1;
        this.renderSearchPanel();
        this.renderTree();
        this.focusSafeContainer();
      };
    }
  }

  private renderBreadcrumbs() {
    this.crumbsEl.empty();
    const root = this.getRootInCurrentGroup();
    if (!root) {
      this.crumbsEl.addClass("is-hidden");
      return;
    }
    const branchRoot = this.getActiveBranchRoot(root);
    if (branchRoot.path === root.path) {
      this.crumbsEl.addClass("is-hidden");
      return;
    }
    this.crumbsEl.removeClass("is-hidden");

    const chain = buildBreadcrumbChain(this.scan, root.path, branchRoot.path);

    chain.forEach((node, idx) => {
      if (idx > 0) {
        this.crumbsEl.createSpan({ cls: "my-sc-crumb-sep", text: "›" });
      }
      const isLast = idx === chain.length - 1;
      const btn = this.crumbsEl.createEl("button", {
        cls: "my-sc-crumb-btn" + (isLast ? " is-current" : ""),
        text: node.name,
        attr: { type: "button" },
      });
      btn.onclick = () => this.setFocusBranch(root, node.path);
    });

    const resetBtn = this.crumbsEl.createEl("button", {
      cls: "my-sc-crumb-reset",
      text: "Весь проект",
      attr: { type: "button", title: "Вернуться ко всему дереву проекта" },
    });
    resetBtn.onclick = () => this.setFocusBranch(root, root.path);
  }

  private openForm(parentPath: string, focusTitle = true) {
    const root = this.getRootInCurrentGroup();
    if (!root) return;
    this.form.open = true;
    this.form.type =
      this.plugin.settings.lastUsedMaterialType ||
      this.scan.schema.typeCounts[0]?.name ||
      "Термин";
    this.form.parentPath = this.scan.byPath.has(parentPath) ? parentPath : root.path;
    this.renderBar();
    this.renderForm(focusTitle);
    this.formEl.querySelector(".my-sc-add-box")?.addClass("is-entering");
  }

  private renderForm(focusTitle = false) {
    this.formEl.empty();
    if (!this.form.open) return;
    const root = this.getRootInCurrentGroup();
    if (!root) return;

    const box = this.formEl.createDiv("my-sc-add-box");
    const topRow = box.createDiv("my-sc-form-row");

    const typeWrap = topRow.createDiv("my-sc-type-input-wrap");
    const typeInp = typeWrap.createEl("input", {
      cls: "my-sc-type-input",
      attr: {
        type: "text",
        list: "my-sc-types-datalist",
        placeholder: "Тип (Термин, Расчёт, Схема…)",
        title: "Впиши любой тип или выбери из подсказок",
      },
    });
    typeInp.value = this.form.type;

    const caseHintEl = box.createDiv("my-sc-case-hint is-hidden");

    const updateCaseHint = () => {
      const existingCase = findCaseInsensitiveTypeMatch(
        typeInp.value,
        this.scan.schema.typeCounts
      );
      caseHintEl.empty();
      if (!existingCase) {
        caseHintEl.addClass("is-hidden");
        return;
      }
      caseHintEl.removeClass("is-hidden");
      caseHintEl.createSpan({
        text: `В базе уже есть тип «${existingCase}». `,
      });
      const useExistingBtn = caseHintEl.createEl("button", {
        cls: "my-sc-inline-link-btn",
        text: `Использовать «${existingCase}»`,
        attr: { type: "button" },
      });
      useExistingBtn.onclick = () => {
        this.form.type = existingCase;
        typeInp.value = existingCase;
        caseHintEl.addClass("is-hidden");
      };
    };

    typeInp.oninput = () => {
      this.form.type = typeInp.value;
      updateCaseHint();
    };
    updateCaseHint();

    const titleInp = topRow.createEl("input", {
      cls: "my-sc-input",
      attr: { type: "text", placeholder: "Название темы или материала…" },
    });
    titleInp.value = this.form.title;
    titleInp.oninput = () => {
      this.form.title = titleInp.value;
    };

    const parentSel = topRow.createEl("select", {
      cls: "dropdown my-sc-parent-select",
    });
    parentSel.createEl("option", { text: `↳ ${root.name}`, value: root.path });
    const allTree = buildNestedFilteredTree(root, this.scan, null, {
      query: "",
      mode: "all",
      typeFilter: "",
    });
    for (const n of flattenVisibleTree(allTree)) {
      parentSel.createEl("option", {
        text: `${"\u00A0\u00A0".repeat(n.depth + 1)}↳ ${n.item.name}`,
        value: n.item.path,
      });
    }
    parentSel.value = this.form.parentPath;
    if (parentSel.value !== this.form.parentPath) {
      this.form.parentPath = root.path;
      parentSel.value = root.path;
    }
    parentSel.onchange = () => {
      this.form.parentPath = parentSel.value;
    };

    const dateInp = topRow.createEl("input", {
      cls: "my-sc-date",
      attr: { type: "date", title: "Дедлайн (необязательно)" },
    });
    dateInp.value = this.form.deadline;
    dateInp.onchange = () => {
      this.form.deadline = dateInp.value;
    };

    if (this.scan.schema.typeCounts.length > 0) {
      const quickTypesRow = box.createDiv("my-sc-quick-types-row");
      quickTypesRow.createSpan({
        cls: "my-sc-quick-types-label",
        text: "Частые типы:",
      });
      for (const tc of this.scan.schema.typeCounts.slice(0, 7)) {
        const chip = quickTypesRow.createEl("button", {
          cls:
            "my-sc-filter-chip" +
            (normalizeSpaces(this.form.type).toLowerCase() ===
            tc.name.toLowerCase()
              ? " is-active"
              : ""),
          text: `${tc.name} (${tc.count})`,
          attr: { type: "button" },
        });
        chip.onclick = () => {
          this.form.type = tc.name;
          this.renderForm(false);
        };
      }
    }

    const propsSection = box.createDiv("my-sc-props-section");
    for (let i = 0; i < this.form.customProps.length; i++) {
      const draft = this.form.customProps[i];
      if (!draft) continue;
      const pRow = propsSection.createDiv("my-sc-prop-row");
      const valDatalistId = `my-sc-val-dl-${draft.id}`;

      const keyInp = pRow.createEl("input", {
        cls: "my-sc-prop-key",
        attr: {
          type: "text",
          list: "my-sc-prop-keys-datalist",
          placeholder: "Ключ (напр. источник, автор)…",
        },
      });
      keyInp.value = draft.key;

      const valDl = pRow.createEl("datalist", { attr: { id: valDatalistId } });
      const populateValSuggestions = (kName: string) => {
        valDl.empty();
        const vals = this.scan.schema.valuesByKey.get(normalizeSpaces(kName)) ?? [];
        for (const v of vals.slice(0, 25)) {
          valDl.createEl("option", { attr: { value: v } });
        }
      };
      populateValSuggestions(draft.key);

      keyInp.oninput = () => {
        draft.key = keyInp.value;
        populateValSuggestions(draft.key);
      };

      const valInp = pRow.createEl("input", {
        cls: "my-sc-prop-val",
        attr: {
          type: "text",
          list: valDatalistId,
          placeholder:
            draft.kind === "list"
              ? "Значения через запятую…"
              : "Значение (дата ГГГГ-ММ-ДД, число или текст)…",
        },
      });
      valInp.value = draft.value;
      valInp.oninput = () => {
        draft.value = valInp.value;
      };

      const kindSel = pRow.createEl("select", {
        cls: "dropdown my-sc-prop-kind",
        attr: { title: "Формат записи в YAML" },
      });
      kindSel.createEl("option", { text: "Значение", value: "auto" });
      kindSel.createEl("option", { text: "Список (,)", value: "list" });
      kindSel.value = draft.kind;
      kindSel.onchange = () => {
        draft.kind = kindSel.value === "list" ? "list" : "auto";
        valInp.placeholder =
          draft.kind === "list"
            ? "Значения через запятую…"
            : "Значение (дата ГГГГ-ММ-ДД, число или текст)…";
      };

      this.iconBtn(pRow, "x", "Удалить свойство", () => {
        this.form.customProps.splice(i, 1);
        this.renderForm(false);
      });
    }

    const actionsRow = box.createDiv("my-sc-form-actions");
    const addPropBtn = actionsRow.createEl("button", {
      cls: "my-sc-chip-btn",
      text: "Добавить свойство",
      attr: { type: "button" },
    });
    addPropBtn.onclick = () => {
      this.form.customProps.push({
        id: String(Date.now() + Math.random()),
        key: "",
        value: "",
        kind: "auto",
      });
      this.renderForm(false);
    };

    const createBtn = actionsRow.createEl("button", {
      cls: "mod-cta my-sc-action-btn",
      text: "Создать материал",
      attr: { type: "button" },
    });

    const submit = async () => {
      createBtn.disabled = true;
      const ok = await this.createNote(root);
      if (ok) {
        this.form.title = "";
        this.form.deadline = "";
        this.form.customProps = [];
        this.renderForm(true);
      } else {
        createBtn.disabled = false;
      }
    };
    createBtn.onclick = () => void submit();
    titleInp.addEventListener("keydown", (e: KeyboardEvent) => {
      if (e.key === "Enter" && !e.isComposing) void submit();
    });
    if (focusTitle) {
      window.setTimeout(() => titleInp.focus(), 20);
    }
  }

  private async createNote(root: StudyNoteItem): Promise<boolean> {
    const cleanTitle = normalizeSpaces(this.form.title).replace(/[\\/:*?"<>|]/g, "");
    if (!cleanTitle) {
      new Notice("Введи название материала.");
      return false;
    }

    const normalizedType = normalizeSpaces(this.form.type) || "Материал";

    const seenKeysLower = new Set<string>([
      KEYS.area.toLowerCase(),
      KEYS.type.toLowerCase(),
      KEYS.parent.toLowerCase(),
    ]);
    if (root.discipline) seenKeysLower.add(KEYS.discipline.toLowerCase());
    if (/^\d{4}-\d{2}-\d{2}$/.test(this.form.deadline)) {
      seenKeysLower.add(KEYS.deadline.toLowerCase());
    }

    const preparedProps: Array<{ key: string; serialized: string }> = [];

    for (const draft of this.form.customProps) {
      const k = normalizeSpaces(draft.key);
      const parsedVal = parseCustomPropertyValue(draft.value, draft.kind);
      if (!k && parsedVal === undefined) continue;
      if (!k) {
        new Notice("Укажи имя ключа для добавленного свойства или удали пустую строку.");
        return false;
      }
      if (isReservedYamlKey(k)) {
        new Notice(
          `Ключ «${k}» задаётся основными полями формы. Удали его из дополнительных свойств.`
        );
        return false;
      }
      const kLow = k.toLowerCase();
      if (seenKeysLower.has(kLow)) {
        new Notice(`Дублирующийся ключ свойства «${k}». Оставь только одно поле.`);
        return false;
      }
      if (parsedVal === undefined) continue;

      seenKeysLower.add(kLow);
      preparedProps.push({
        key: k,
        serialized: serializeYamlScalarOrList(parsedVal),
      });
    }

    const folder = normalizePath(this.plugin.settings.newNoteFolder || "Знания");
    const targetPath = normalizePath(`${folder}/${cleanTitle}.md`);
    if (this.app.vault.getAbstractFileByPath(targetPath)) {
      new Notice(`Заметка «${cleanTitle}» уже есть в папке ${folder}.`);
      return false;
    }
    const parent = this.scan.byPath.get(this.form.parentPath) ?? root;

    try {
      await ensureFolder(this.app, folder);
      const link = this.app.metadataCache.fileToLinktext(parent.file, targetPath);
      const lines = [
        "---",
        `${KEYS.area}: ${JSON.stringify(this.plugin.settings.studyAreaName)}`,
        `${KEYS.type}: ${JSON.stringify(normalizedType)}`,
        `${KEYS.parent}: ${JSON.stringify(`[[${link}]]`)}`,
      ];
      if (root.discipline) {
        lines.push(`${KEYS.discipline}: ${JSON.stringify(root.discipline)}`);
      }
      if (/^\d{4}-\d{2}-\d{2}$/.test(this.form.deadline)) {
        lines.push(`${KEYS.deadline}: ${JSON.stringify(this.form.deadline)}`);
      }
      for (const prop of preparedProps) {
        lines.push(`${prop.key}: ${prop.serialized}`);
      }
      lines.push("---", "", `# ${cleanTitle}`, "", "");
      await this.app.vault.create(targetPath, lines.join("\n"));

      this.plugin.settings.lastUsedMaterialType = normalizedType;
      this.expanded.add(parent.path);
      this.saveExpanded();
      new Notice(`Создано: «${cleanTitle}» (${normalizedType})`);
      return true;
    } catch (err: unknown) {
      new Notice("Ошибка создания: " + (err instanceof Error ? err.message : String(err)));
      return false;
    }
  }

  private renderDeadline(parent: HTMLElement, item: StudyNoteItem) {
    if (!item.deadline) return;
    const tone = getDeadlineTone(item);
    const cls =
      tone === "overdue"
        ? " is-overdue"
        : tone === "soon"
          ? " is-soon"
          : " is-far";
    parent.createSpan({
      cls: "my-sc-deadline" + cls,
      text: formatDeadlineBadge(item),
      attr: {
        title: formatFullDateTooltip(item.deadline),
      },
    });
  }

  private renderHighlightedTitle(container: HTMLElement, text: string, query: string) {
    const cleanQ = query
      .trim()
      .split(/\s+/)
      .filter((t) => !t.startsWith("#"))[0];
    if (!cleanQ) {
      container.setText(text);
      return;
    }
    const lowerText = text.toLowerCase();
    const lowerQ = cleanQ.toLowerCase();
    const idx = lowerText.indexOf(lowerQ);
    if (idx === -1) {
      container.setText(text);
      return;
    }
    if (idx > 0) container.appendText(text.slice(0, idx));
    container.createEl("mark", {
      cls: "my-sc-hl",
      text: text.slice(idx, idx + cleanQ.length),
    });
    if (idx + cleanQ.length < text.length) {
      container.appendText(text.slice(idx + cleanQ.length));
    }
  }

  private openNote(path: string) {

  }

  private renderTree() {
    const scrollTop = this.treeEl.scrollTop;
    this.treeEl.empty();
    this.rowElementByPath.clear();
    const root = this.getRootInCurrentGroup();
    if (!root) return;

    const branchRoot = this.getActiveBranchRoot(root);
    const nestedNodes = buildNestedFilteredTree(
      branchRoot,
      this.scan,
      this.expanded,
      {
        query: this.searchQuery,
        mode: this.filterMode,
        typeFilter: this.typeFilter,
      }
    );
    this.currentVisibleFlat = flattenVisibleTree(nestedNodes);

    if (nestedNodes.length === 0) {
      const hasFilters =
        this.searchQuery.trim().length > 0 ||
        this.filterMode !== "all" ||
        this.typeFilter.length > 0;
      this.treeEl.createDiv({
        cls: "my-sc-empty",
        text: hasFilters
          ? "Ничего не найдено."
          : "Здесь пока пусто. Добавь материал кнопкой «+».",
      });
      this.pendingOpenPath = null;
      this.syncExpandToggle();
      return;
    }

    const listEl = this.treeEl.createDiv("my-sc-tree-list");
    for (const node of nestedNodes) {
      this.renderNodeGroup(listEl, node, root);
    }

    if (this.activeRowIndex >= 0) {
      this.highlightActiveRow();
    }
    this.treeEl.scrollTop = scrollTop;

    if (this.pendingOpenPath) {
      const openPath = this.pendingOpenPath;
      this.pendingOpenPath = null;
      const kids = this.getChildrenContainer(openPath);
      if (kids) animateOpen(kids);
      if (!prefersReducedMotion()) {
        const btn = this.rowElementByPath
          .get(openPath)
          ?.querySelector<HTMLElement>(".my-sc-toggle-btn");
        if (btn) {
          btn.removeClass("is-open");
          void btn.offsetWidth;
          btn.addClass("is-open");
        }
      }
    }

    this.syncExpandToggle();
  }

  private renderNodeGroup(
    parentContainer: HTMLElement,
    node: TreeNode,
    projectRoot: StudyNoteItem
  ) {
    const item = node.item;
    const depthSlot = node.depth % 5;

    const groupEl = parentContainer.createDiv({
      cls:
        "my-sc-node-group" +
        ` is-depth-${depthSlot}` +
        (node.hasChildren ? " is-branch" : " is-leaf") +
        (node.isExpanded ? " is-expanded" : " is-collapsed"),
    });

    const row = groupEl.createDiv({
      cls:
        "my-sc-tree-row" +
        (node.hasChildren ? " is-parent-row" : "") +
        (node.isExpanded ? " is-open-parent" : "") +
        (node.isDirectMatch ? " is-match" : "") +
        (this.freshPaths.has(item.path) ? " is-new" : ""),
    });
    this.rowElementByPath.set(item.path, row);
    row.oncontextmenu = (e) => {
      e.preventDefault();
      this.openRowMenu(e, item, projectRoot);
    };

    const left = row.createDiv("my-sc-tree-left");

    if (node.hasChildren) {
      const toggleBtn = left.createEl("button", {
        cls: "my-sc-toggle-btn" + (node.isExpanded ? " is-open" : ""),
        attr: {
          type: "button",
          title: node.isExpanded ? "Свернуть ветку" : "Развернуть ветку",
          "aria-label": node.isExpanded ? "Свернуть ветку" : "Развернуть ветку",
          "aria-expanded": String(node.isExpanded),
        },
      });
      setIcon(toggleBtn, "chevron-right");
      toggleBtn.onclick = (e) => {
        e.stopPropagation();
        this.toggleBranch(item.path);
      };
    } else {
      const dotWrap = left.createSpan("my-sc-leaf-marker");
      dotWrap.createSpan("my-sc-leaf-dot");
    }

    const title = left.createSpan({
      cls: "my-sc-node-title",
      attr: { title: `${item.name} (${item.path})` },
    });
    this.renderHighlightedTitle(title, item.name, this.searchQuery);
    title.onclick = () => this.openNote(item.path);

    if (item.parentPaths.length > 1) {
      left.createSpan({
        cls: "my-sc-multi-parent",
        text: `×${item.parentPaths.length}`,
        attr: { title: `Привязана к ${item.parentPaths.length} родителям` },
      });
    }

    const right = row.createDiv("my-sc-tree-right");
    this.renderDeadline(right, item);

    const meta = getWorkMeta(item.noteType, item.icon);
    const kind = right.createEl("button", {
      cls: "my-sc-type-badge",
      attr: {
        type: "button",
        title: `${item.noteType || "Без типа"} · нажми, чтобы изменить`,
      },
    });
    kind.createSpan({ cls: "my-sc-type-icon", text: meta.icon });
    kind.createSpan({ cls: "my-sc-type-name", text: item.noteType || "без типа" });
    kind.style.setProperty("--badge-color", meta.color);
    kind.onclick = (e) => {
      e.stopPropagation();
      if (this.editingNodePath === item.path && this.editingNodeMode === "type") {
        this.editingNodePath = null;
        this.editingNodeMode = null;
        this.renderTree();
      } else {
        this.startNodeEditing(item.path, "type");
      }
    };

    if (node.hasChildren) {
      const countPill = right.createEl("button", {
        cls:
          "my-sc-subcount-pill" +
          (node.isExpanded ? " is-expanded" : " is-collapsed"),
        text: `${node.descendantCount}`,
        attr: {
          type: "button",
          title: node.isExpanded
            ? `Вложено: ${node.descendantCount} (нажми, чтобы свернуть)`
            : `Скрыто вложенных: ${node.descendantCount} (нажми, чтобы раскрыть)`,
        },
      });
      countPill.onclick = (e) => {
        e.stopPropagation();
        this.toggleBranch(item.path);
      };
    }

    const moreBtn = right.createEl("button", {
      cls: "my-sc-row-more",
      attr: { type: "button", title: "Действия", "aria-label": "Действия с материалом" },
    });
    setIcon(moreBtn, "more-horizontal");
    moreBtn.onclick = (e) => {
      e.stopPropagation();
      this.openRowMenu(e, item, projectRoot);
    };

    if (this.editingNodePath === item.path && this.editingNodeMode) {
      this.renderInlineNodeEditor(groupEl, item);
    }

    if (node.isExpanded && node.children.length > 0) {
      const childrenWrap = groupEl.createDiv("my-sc-children-container");
      for (const child of node.children) {
        this.renderNodeGroup(childrenWrap, child, projectRoot);
      }
    }
  }

  private renderInlineNodeEditor(parentGroup: HTMLElement, item: StudyNoteItem) {
    const editorBox = parentGroup.createDiv("my-sc-inline-editor");

    if (this.editingNodeMode === "type") {
      editorBox.createSpan({ cls: "my-sc-inline-label", text: "Тип:" });
      const inp = editorBox.createEl("input", {
        cls: "my-sc-type-input",
        attr: {
          type: "text",
          list: "my-sc-types-datalist",
          value: item.noteType,
          placeholder: "Новый тип…",
        },
      });
      const saveBtn = editorBox.createEl("button", {
        cls: "mod-cta my-sc-chip-btn",
        text: "Сохранить",
        attr: { type: "button" },
      });
      const cancelBtn = editorBox.createEl("button", {
        cls: "my-sc-chip-btn",
        text: "Отмена",
        attr: { type: "button" },
      });

      const applyType = async () => {
        const nextType = normalizeSpaces(inp.value);
        if (!nextType) {
          new Notice("Введи название типа.");
          return;
        }
        try {
          await this.app.fileManager.processFrontMatter(
            item.file,
            (fm: Record<string, unknown>) => {
              fm[KEYS.type] = nextType;
            }
          );
          this.plugin.settings.lastUsedMaterialType = nextType;
          this.plugin.requestSaveSettings();
          this.editingNodePath = null;
          this.editingNodeMode = null;
          new Notice(`Тип «${item.name}» → ${nextType}`);
        } catch (err: unknown) {
          new Notice(
            "Ошибка записи YAML: " +
              (err instanceof Error ? err.message : String(err))
          );
        }
      };

      saveBtn.onclick = () => void applyType();
      cancelBtn.onclick = () => {
        this.editingNodePath = null;
        this.editingNodeMode = null;
        this.renderTree();
        this.focusSafeContainer();
      };
      inp.addEventListener("keydown", (e: KeyboardEvent) => {
        if (e.key === "Enter") void applyType();
      });
      window.setTimeout(() => inp.focus(), 20);
      return;
    }

    editorBox.createSpan({ cls: "my-sc-inline-label", text: "Свойство YAML:" });
    const dlId = `my-sc-inline-val-${Date.now()}`;
    const keyInp = editorBox.createEl("input", {
      cls: "my-sc-prop-key",
      attr: {
        type: "text",
        list: "my-sc-prop-keys-datalist",
        placeholder: "Ключ (напр. источник)…",
      },
    });
    const valDl = editorBox.createEl("datalist", { attr: { id: dlId } });
    keyInp.oninput = () => {
      valDl.empty();
      const vals =
        this.scan.schema.valuesByKey.get(normalizeSpaces(keyInp.value)) ?? [];
      for (const v of vals.slice(0, 25)) {
        valDl.createEl("option", { attr: { value: v } });
      }
    };

    const valInp = editorBox.createEl("input", {
      cls: "my-sc-prop-val",
      attr: {
        type: "text",
        list: dlId,
        placeholder: "Значение (пусто = удалить ключ)…",
      },
    });

    const kindSel = editorBox.createEl("select", {
      cls: "dropdown my-sc-prop-kind",
    });
    kindSel.createEl("option", { text: "Значение", value: "auto" });
    kindSel.createEl("option", { text: "Список (,)", value: "list" });

    const saveBtn = editorBox.createEl("button", {
      cls: "mod-cta my-sc-chip-btn",
      text: "Записать",
      attr: { type: "button" },
    });
    const cancelBtn = editorBox.createEl("button", {
      cls: "my-sc-chip-btn",
      text: "Отмена",
      attr: { type: "button" },
    });

    const applyProp = async () => {
      const k = normalizeSpaces(keyInp.value);
      if (!k) {
        new Notice("Укажи ключ свойства.");
        return;
      }
      if (isReservedYamlKey(k)) {
        new Notice(
          `Служебный ключ «${k}» меняется через кнопку типа или смены родителя.`
        );
        return;
      }
      const kind = kindSel.value === "list" ? "list" : "auto";
      const parsedVal = parseCustomPropertyValue(valInp.value, kind);
      try {
        await this.app.fileManager.processFrontMatter(
          item.file,
          (fm: Record<string, unknown>) => {
            if (parsedVal === undefined) {
              delete fm[k];
            } else {
              fm[k] = parsedVal;
            }
          }
        );
        this.editingNodePath = null;
        this.editingNodeMode = null;
        new Notice(
          parsedVal === undefined
            ? `Свойство «${k}» удалено из «${item.name}»`
            : `Записано «${k}» в «${item.name}»`
        );
      } catch (err: unknown) {
        new Notice(
          "Ошибка записи YAML: " +
            (err instanceof Error ? err.message : String(err))
        );
      }
    };

    saveBtn.onclick = () => void applyProp();
    cancelBtn.onclick = () => {
      this.editingNodePath = null;
      this.editingNodeMode = null;
      this.renderTree();
      this.focusSafeContainer();
    };
    window.setTimeout(() => keyInp.focus(), 20);
  }

  private openMovePicker(item: StudyNoteItem) {
    const forbidden = collectDescendants(item.path, this.scan);
    forbidden.add(item.path);
    const candidates = [...this.scan.byPath.values()]
      .filter((c) => !forbidden.has(c.path))
      .sort(byName);
    new ParentPickerModal(this.app, candidates, (target) => {
      void this.setParent(item, target);
    }).open();
  }

  private async setParent(item: StudyNoteItem, target: StudyNoteItem) {
    try {
      const link = this.app.metadataCache.fileToLinktext(target.file, item.path);
      await this.app.fileManager.processFrontMatter(
        item.file,
        (fm: Record<string, unknown>) => {
          fm[KEYS.parent] = `[[${link}]]`;
        }
      );
      this.expanded.add(target.path);
      this.saveExpanded();
      new Notice(`«${item.name}» → «${target.name}»`);
    } catch (err: unknown) {
      new Notice(
        "Не удалось сменить родителя: " +
          (err instanceof Error ? err.message : String(err))
      );
    }
  }
}

function migrateSettings(raw: unknown): StudyCompilerSettings {
  const s =
    raw && typeof raw === "object"
      ? (raw as Partial<StudyCompilerSettings> & { showCompletedWorks?: boolean })
      : {};
  const strList = (v: unknown, fallback: string[]): string[] =>
    Array.isArray(v)
      ? v.filter((x): x is string => typeof x === "string" && x.trim().length > 0)
      : [...fallback];

  const focusedBranchByWork: Record<string, string> = {};
  if (s.focusedBranchByWork && typeof s.focusedBranchByWork === "object") {
    for (const [k, v] of Object.entries(s.focusedBranchByWork as Record<string, unknown>)) {
      if (typeof v === "string" && v.trim()) focusedBranchByWork[k] = v;
    }
  }

  let projectGroupTab: ProjectGroupTab = "active";
  if (s.projectGroupTab === "active" || s.projectGroupTab === "completed") {
    projectGroupTab = s.projectGroupTab;
  }

  return {
    studyAreaName:
      typeof s.studyAreaName === "string" && s.studyAreaName.trim()
        ? s.studyAreaName.trim()
        : DEFAULT_SETTINGS.studyAreaName,
    newNoteFolder:
      typeof s.newNoteFolder === "string" && s.newNoteFolder.trim()
        ? s.newNoteFolder.trim()
        : DEFAULT_SETTINGS.newNoteFolder,
    projectGroupTab,
    lastUsedMaterialType:
      typeof s.lastUsedMaterialType === "string" && s.lastUsedMaterialType.trim()
        ? s.lastUsedMaterialType.trim()
        : DEFAULT_SETTINGS.lastUsedMaterialType,
    excludedFolders: strList(s.excludedFolders, DEFAULT_SETTINGS.excludedFolders),
    excludedTitlePrefixes: strList(
      s.excludedTitlePrefixes,
      DEFAULT_SETTINGS.excludedTitlePrefixes
    ),
    excludedTags: strList(s.excludedTags, DEFAULT_SETTINGS.excludedTags),
    expandedPaths: strList(s.expandedPaths, []),
    lastWorkPath: typeof s.lastWorkPath === "string" ? s.lastWorkPath : "",
    focusedBranchByWork,
  };
}

export default class MyStudyCompilerPlugin extends Plugin {
  settings!: StudyCompilerSettings;
  private saveTimer: number | null = null;

  async onload() {
    this.settings = migrateSettings(await this.loadData());
    this.addSettingTab(new StudyCompilerSettingTab(this.app, this));

    this.addRibbonIcon("folder-tree", "Открыть проекты и подпроекты", () => {
      this.openCompiler();
    });

    this.registerObsidianProtocolHandler("my-compiler", (params) => {
      this.openCompiler({
        rootPath: params.project || undefined,
        addUnder: params.parent || undefined,
      });
    });

    this.addCommand({
      id: "open-study-compiler",
      name: "Открыть проекты и подпроекты",
      callback: () => this.openCompiler(),
    });

    this.addCommand({
      id: "new-subproject-here",
      name: "Новый подпроект в текущей заметке",
      checkCallback: (checking: boolean) => {
        const file = this.app.workspace.getActiveFile();
        if (!file) return false;
        if (!checking) this.openCompiler({ addUnder: file.path });
        return true;
      },
    });
  }

  openCompiler(options: OpenOptions = {}) {
    new StudyCompilerModal(this.app, this, options).open();
  }

  requestSaveSettings() {
    if (this.saveTimer !== null) window.clearTimeout(this.saveTimer);
    this.saveTimer = window.setTimeout(() => {
      this.saveTimer = null;
      void this.saveData(this.settings);
    }, 300);
  }

  onunload() {
    if (this.saveTimer !== null) {
      window.clearTimeout(this.saveTimer);
      this.saveTimer = null;
    }
  }
}
